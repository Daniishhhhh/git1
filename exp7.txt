from PIL import Image

# HIDE
def hide():
    img_path = input("Enter image file: ")
    msg = input("Enter message: ")

    msg += "#####"
    bits = ''.join(format(ord(c),'08b') for c in msg)

    img = Image.open(img_path)
    data = list(img.getdata())

    new = []
    index = 0

    for pixel in data:
        r,g,b = pixel
        if index < len(bits):
            r = (r & ~1) | int(bits[index])
            index += 1
        new.append((r,g,b))

    img.putdata(new)
    img.save("stego.png")
    print("Message hidden in stego.png")


# EXTRACT
def extract():
    img_path = input("Enter stego image: ")

    img = Image.open(img_path)
    data = list(img.getdata())

    bits = ""
    for r,g,b in data:
        bits += str(r & 1)

    msg = ""
    for i in range(0,len(bits),8):
        byte = bits[i:i+8]
        if len(byte) < 8:
            break

        ch = chr(int(byte,2))
        msg += ch

        if msg.endswith("#####"):
            print("Hidden Message:", msg[:-5])
            return

    print("No message found")


# MENU
print("1. Hide Message")
print("2. Extract Message")

choice = int(input("Enter choice: "))

if choice == 1:
    hide()
elif choice == 2:
    extract()
else:
    print("Invalid choice")
